import { useState, useEffect, useRef } from "react";

function App() {
  const [loading, setLoading] = useState(true);
  const [activeMemory, setActiveMemory] = useState(null);

  // Generate hearts once
  const heartsRef = useRef(
    [...Array(50)].map((_, i) => ({
      id: i,
      left: Math.random() * 100,
      delay: Math.random() * 5,
      duration: 4 + Math.random() * 4,
      size: 12 + Math.random() * 20,
      color: Math.random() > 0.5 ? "#f472b6" : "#ec4899",
    }))
  );

  const memories = [
    {
      id: 1,
      icon: "📸",
      image: "https://picsum.photos/400/300?random=1",
      text: "The day I realized you were different."
    },
    {
      id: 2,
      icon: "🌙",
      image: "https://picsum.photos/400/300?random=2",
      text: "Late night talks that meant everything."
    },
    {
      id: 3,
      icon: "💌",
      image: "https://picsum.photos/400/300?random=3",
      text: "Unsent messages that held my heart."
    }
  ];

  useEffect(() => {
    const timer = setTimeout(() => {
      setLoading(false);
    }, 7000);
    return () => clearTimeout(timer);
  }, []);

  // 🌸 LOADING SCREEN
  if (loading) {
    return (
      <div className="min-h-screen bg-[#f5ede3] relative overflow-hidden flex items-center justify-center">

        <div className="absolute w-72 h-72 bg-pink-300 rounded-full blur-3xl opacity-30 slow-blob top-10 left-10"></div>
        <div className="absolute w-80 h-80 bg-pink-400 rounded-full blur-3xl opacity-20 slow-blob bottom-10 right-10"></div>
        <div className="absolute w-60 h-60 bg-pink-200 rounded-full blur-3xl opacity-30 slow-blob top-1/3 right-1/4"></div>

        {heartsRef.current.map((heart) => (
          <span
            key={heart.id}
            className="absolute float-heart"
            style={{
              left: `${heart.left}%`,
              bottom: "-20px",
              fontSize: `${heart.size}px`,
              animationDuration: `${heart.duration}s`,
              animationDelay: `${heart.delay}s`,
              color: heart.color,
            }}
          >
            ❤
          </span>
        ))}

        <h1 className="relative text-3xl md:text-4xl font-semibold text-pink-500 flex items-center gap-2 z-10">
          Wait, My love's Loading
          <span className="dot-1">.</span>
          <span className="dot-2">.</span>
          <span className="dot-3">.</span>
        </h1>
      </div>
    );
  }

  // 💖 MAIN PAGE
  return (
    <div className="min-h-screen bg-[#f5ede3] relative overflow-x-hidden">

      {/* LEFT SIDEBAR */}
      <div className="fixed left-0 top-0 h-full w-20 bg-white/30 backdrop-blur-md flex flex-col items-center justify-center gap-10 border-r border-pink-200 z-20">
        {memories.map((memory) => (
          <div
            key={memory.id}
            onMouseEnter={() => setActiveMemory(memory)}
            onMouseLeave={() => setActiveMemory(null)}
            className="text-2xl cursor-pointer hover:scale-125 transition duration-300"
          >
            {memory.icon}
          </div>
        ))}
      </div>

      {/* MEMORY PREVIEW */}
      {activeMemory && (
        <div className="fixed left-28 top-1/2 -translate-y-1/2 bg-white rounded-xl shadow-2xl p-4 w-80 animate-fadeIn z-30">
          <img
            src={activeMemory.image}
            alt="memory"
            className="rounded-lg mb-3 w-full h-48 object-cover"
          />
          <p className="text-gray-700 text-sm text-center">
            {activeMemory.text}
          </p>
        </div>
      )}

      {/* Top Right Navigation */}
      <div className="absolute top-6 right-8 flex gap-8 text-pink-500 font-medium z-20">
        <button className="hover:text-pink-600 transition duration-300">
          Games
        </button>
        <button className="hover:text-pink-600 transition duration-300">
          Gallery
        </button>
      </div>

      {/* Hero Section */}
      <div className="min-h-[80vh] flex flex-col items-center justify-center text-center px-4">
        <h1 className="text-5xl md:text-6xl font-bold text-pink-500 mb-6">
          Hey Jerosh Chris Rohan!!
        </h1>

        <p className="text-gray-600 text-lg md:text-xl">
          [Warning: This page contains feelings.]
        </p>
      </div>

      {/* Bottom Right Heart Button */}
      <div className="absolute bottom-10 right-10 z-20">
        <button className="w-14 h-14 bg-pink-400 rounded-full flex items-center justify-center text-white text-xl hover:bg-pink-500 transition duration-300 shadow-lg">
          ❤
        </button>
      </div>

      {/* PREMIUM FOOTER */}
      <footer className="bg-gradient-to-br from-[#0f172a] to-[#1e293b] text-white mt-24 px-6 py-20">
        <div className="max-w-6xl mx-auto">
          <div className="bg-white/5 backdrop-blur-lg rounded-2xl p-12 shadow-2xl">

            <div className="text-center mb-16">
              <h2 className="text-2xl md:text-3xl font-semibold text-white mb-4">
                Created exclusively for JCRohan
              </h2>
              <p className="text-gray-400 text-sm md:text-base">
                A private experience created just for you.
              </p>
            </div>

            <div className="grid md:grid-cols-3 gap-12">
              <div>
                <h3 className="text-lg font-semibold mb-6">Products</h3>
                <ul className="space-y-3 text-gray-300">
                  <li>Love 2.0</li>
                  <li>Butterflies.exe</li>
                  <li>Long Nights</li>
                  <li>Future Memories</li>
                </ul>
              </div>

              <div>
                <h3 className="text-lg font-semibold mb-6">Info</h3>
                <ul className="space-y-3 text-gray-300">
                  <li>Privacy: You are safe here</li>
                  <li>Terms: No ghosting</li>
                  <li>Support: Always here</li>
                  <li>Version: Valentine 2026</li>
                </ul>
              </div>

              <div>
                <h3 className="text-lg font-semibold mb-6">Legals</h3>
                <ul className="space-y-3 text-gray-300">
                  <li>Terms & Feelings</li>
                  <li>Privacy Policy</li>
                  <li>No Ghosting Clause</li>
                  <li>License to Love</li>
                </ul>
              </div>
            </div>

            <div className="border-t border-white/10 my-12"></div>

            <div className="flex flex-col md:flex-row justify-between items-center text-gray-400 text-sm">
              <p>
                <strong>
                  Made with Keyboard, VSCode, React+Vite, MongoDB, Tailwind and definitely with love.
                </strong>
              </p>

              <div className="flex gap-2 mt-4 md:mt-0 items-center">
                <span className="text-[#f9a8d4] font-medium">
                  AnnonceM
                </span>
                <span className="text-white">
                  © 2026
                </span>
              </div>
            </div>

          </div>
        </div>
      </footer>

    </div>
  );
}

export default App;
